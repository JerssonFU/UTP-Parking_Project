CREATE SEQUENCE IF NOT EXISTS public.rol_id_rol_seq
    INCREMENT 1
    START 1
    MINVALUE 1
    MAXVALUE 2147483647
    CACHE 1;

CREATE SEQUENCE IF NOT EXISTS public."USUARIO_C_PERL_CODIGO_seq"
    INCREMENT 1
    START 1
    MINVALUE 1
    MAXVALUE 2147483647
    CACHE 1;

CREATE TABLE IF NOT EXISTS public.rolesusuario
(
    id_rol integer NOT NULL DEFAULT nextval('public.rol_id_rol_seq'::regclass),
    descripcion_rol character varying(255),
    alias_rol character varying(255),
    CONSTRAINT rol_pkey PRIMARY KEY (id_rol)
);

CREATE TABLE IF NOT EXISTS public.usuario
(
    "C_PERL_CODIGO" integer NOT NULL DEFAULT nextval('public."USUARIO_C_PERL_CODIGO_seq"'::regclass),
    "N_MST_LOGIN" character varying(20),
    "N_MST_NOMBRE" character varying(20),
    "N_MST_APEPATERNO" character varying(20),
    "N_MST_APEMATERNO" character varying(20),
    "N_MST_EMAIL" character varying(50),
    "C_SIT_CODIGO" character(1),
    "N_MST_PASSWORD" character varying(100),
    "C_UNO_CODIGO_OFICINA" character varying(20),
    "N_UNO_DESCRIPCION" character varying(20),
    "ID_ROL" integer,
    CONSTRAINT "USUARIO_pkey" PRIMARY KEY ("C_PERL_CODIGO")
);

CREATE TABLE IF NOT EXISTS public.sedes
(
    id_sede integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    cantidad integer,
    direccion character varying(60),
    nombre character varying(50)
);

CREATE TABLE IF NOT EXISTS public.estacionamientos
(
    id_estacionamiento integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    disponible boolean NOT NULL DEFAULT true,
    nombre character varying(50),
    numero integer,
    piso integer,
    id_sede integer
);

CREATE TABLE IF NOT EXISTS public.vehiculos
(
    id_vehiculo integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    activo boolean NOT NULL DEFAULT false,
    aprobado boolean NOT NULL DEFAULT false,
    categoria character varying(20),
    placa character varying(15),
    id_usuario integer
);

CREATE TABLE IF NOT EXISTS public.solicitudes
(
    id_solicitud integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    estado character varying(15),
    fecha_respuesta timestamp without time zone,
    fecha_solicitud timestamp without time zone,
    id_usuario integer,
    id_vehiculo integer,
    comentario text,
    id_usuario_sae integer,
    id_sede integer,
    id_estacionamiento integer
);

CREATE TABLE IF NOT EXISTS public.registros
(
    id_registro integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    fecha_ingreso timestamp without time zone,
    fecha_salida timestamp without time zone,
    observacion text,
    id_estacionamiento integer,
    id_usuario integer,
    id_usuario_seguridad integer,
    id_vehiculo integer
);

CREATE TABLE IF NOT EXISTS public.usuario_contacto
(
    id_usuario integer PRIMARY KEY,
    celular character varying(15)
);

CREATE OR REPLACE FUNCTION public.roles_usuario(p_codusuario integer)
RETURNS SETOF public.rolesusuario
LANGUAGE plpgsql
AS $BODY$
BEGIN
    RETURN QUERY
    SELECT r.id_rol, r.descripcion_rol, r.alias_rol
    FROM public.rolesusuario r
    INNER JOIN public.usuario u ON u."ID_ROL" = r.id_rol
    WHERE u."C_PERL_CODIGO" = p_codusuario;
END;
$BODY$;

CREATE OR REPLACE FUNCTION public.login_usuario(p_username character varying)
RETURNS SETOF public.usuario
LANGUAGE plpgsql
AS $BODY$
BEGIN
    RETURN QUERY
    SELECT
        u."C_PERL_CODIGO",
        u."N_MST_LOGIN",
        u."N_MST_NOMBRE",
        u."N_MST_APEPATERNO",
        u."N_MST_APEMATERNO",
        u."N_MST_EMAIL",
        u."C_SIT_CODIGO",
        u."N_MST_PASSWORD",
        u."C_UNO_CODIGO_OFICINA",
        u."N_UNO_DESCRIPCION",
        u."ID_ROL"
    FROM public.usuario u
    WHERE UPPER(TRIM(u."N_MST_LOGIN")) = UPPER(TRIM(p_username))
      AND u."C_SIT_CODIGO" = '1';
END;
$BODY$;

INSERT INTO public.estacionamientos (disponible, nombre, numero, piso, id_sede)
SELECT true,
       'Espacio ' || LPAD(g.numero::text, 2, '0'),
       g.numero,
       1,
       s.id_sede
FROM public.sedes s
CROSS JOIN generate_series(1, 30) AS g(numero)
WHERE UPPER(TRIM(s.nombre)) IN ('LIMA CENTRO', 'AREQUIPA', 'PETIT THOUARS')
  AND NOT EXISTS (
      SELECT 1
      FROM public.estacionamientos e
      WHERE e.id_sede = s.id_sede
        AND e.numero = g.numero
        AND e.piso = 1
  );

UPDATE public.solicitudes solicitud
SET id_sede = sede.id_sede
FROM public.sedes sede
WHERE solicitud.id_sede IS NULL
  AND UPPER(TRIM(sede.nombre)) = 'LIMA CENTRO';

UPDATE public.sedes s
SET cantidad = (
    SELECT COUNT(*)
    FROM public.estacionamientos e
    WHERE e.id_sede = s.id_sede
      AND e.disponible = true
)
WHERE UPPER(TRIM(s.nombre)) IN ('LIMA CENTRO', 'AREQUIPA', 'PETIT THOUARS');
